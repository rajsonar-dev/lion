MERN PRACTICAL 1: REACT COMPONENTS, PROPS, CHILDREN, FORM, useState
===================================================================

This practical demonstrates:
- Function component and Class component
- Props and children
- Controlled form
- Event handling and useState

Create a React Vite project:
npm create vite@latest Practical_01 -- --template react
cd Practical_01
npm install
Replace the contents of src/App.jsx with the code below.
Keep the generated src/main.jsx and index.html files.

----- FILE: src/App.jsx -----

import React, { useState } from 'react';
import './App.css';

function Welcome(props) {
  return (
    <div>
      <h2>{props.title}</h2>
      {props.children}
    </div>
  );
}

class Student extends React.Component {
  render() {
    return <p>Student Name: {this.props.name}</p>;
  }
}

function StudentForm() {
  const [name, setName] = useState('');
  const [message, setMessage] = useState('');

  function handleSubmit(event) {
    event.preventDefault();
    if (name.trim() === '') {
      setMessage('Please enter your name.');
    } else {
      setMessage('Hello, ' + name);
    }
  }

  return (
    <form onSubmit={handleSubmit}>
      <input
        value={name}
        onChange={(event) => setName(event.target.value)}
        placeholder="Enter name"
      />
      <button type="submit">Submit</button>
      <p>{message}</p>
    </form>
  );
}

export default function App() {
  const [count, setCount] = useState(0);

  return (
    <div>
      <h1>React Components Practical</h1>
      <Welcome title="Props and Children">
        <p>This paragraph is passed as children.</p>
      </Welcome>
      <h2>Class Component</h2>
      <Student name="Amit" />
      <h2>Form Component</h2>
      <StudentForm />
      <h2>useState and Event Handling</h2>
      <p>Count: {count}</p>
      <button onClick={() => setCount(count + 1)}>Increase</button>
      <button onClick={() => setCount(count - 1)}>Decrease</button>
    </div>
  );
}

----- FILE: src/App.css (optional) -----

body { font-family: Arial; margin: 30px; }
input, button { padding: 8px; margin: 5px; }

----- OUTPUT -----
A page displays Props/Children, a Class Component, a form and Increase/Decrease buttons.
