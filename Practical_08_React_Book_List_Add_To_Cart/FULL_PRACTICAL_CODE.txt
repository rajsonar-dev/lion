MERN PRACTICAL 8: REACT BOOK LIST WITH COMPONENTS, PROPS, CHILDREN AND STATE
=============================================================================

1) SETUP COMMANDS
-----------------
npm create vite@latest book-list -- --template react
cd book-list
npm install
npm run dev

Inside src, create/replace the files below.

2) FILE: src/Heading.jsx
------------------------
import React from "react";

function Heading() {
  return <h1 className="main-heading">Book Store</h1>;
}

export default Heading;

3) FILE: src/ReadingCard.jsx
----------------------------
import React from "react";

function ReadingCard({ title, category, price, onAdd, children }) {
  return (
    <div className="book-card">
      <h2>{title}</h2>
      <p><b>Category:</b> {category}</p>
      <p><b>Price:</b> ₹{price}</p>

          <div className="extra-details">{children}</div>

      <button onClick={onAdd}>Add to Cart</button>
    </div>
  );
}

export default ReadingCard;

4) FILE: src/App.jsx
--------------------
import React, { useState } from "react";
import Heading from "./Heading";
import ReadingCard from "./ReadingCard";
import "./app.css";

function App() {
  const [cartCount, setCartCount] = useState(0);

  function addToCart() {
    setCartCount(cartCount + 1);
  }

  return (
    <div className="container">
      <Heading />
      <h3>Books added to cart: {cartCount}</h3>

      <div className="book-list">
        <ReadingCard
          title="The Alchemist"
          category="Fiction"
          price={299}
          onAdd={addToCart}
        >
          <p>Availability: In stock</p>
          <p>Rating: 4.5/5</p>
        </ReadingCard>

        <ReadingCard
          title="Wings of Fire"
          category="Biography"
          price={250}
          onAdd={addToCart}
        >
          <p>Availability: In stock</p>
          <p>Rating: 4.7/5</p>
        </ReadingCard>

        <ReadingCard
          title="Learning React"
          category="Technology"
          price={450}
          onAdd={addToCart}
        >
          <p>Availability: Limited stock</p>
          <p>Rating: 4.2/5</p>
        </ReadingCard>
      </div>
    </div>
  );
}

export default App;
