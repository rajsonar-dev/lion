MERN PRACTICAL 4: REACT ROUTER, NAVIGATION, ROUTE AND QUERY PARAMETERS
======================================================================

Create React app:
npm create vite@latest Practical_04 -- --template react
cd Practical_04
npm install
npm install react-router-dom
Replace src/App.jsx with the code below. In src/main.jsx wrap App in BrowserRouter
as shown after the main code.

----- FILE: src/App.jsx -----

import { Link, Routes, Route, useParams, useSearchParams } from 'react-router-dom';

function Home() {
  return <h2>Home Page</h2>;
}

function About() {
  return <h2>About Page</h2>;
}

function Student() {
  const { id } = useParams();
  const [searchParams] = useSearchParams();
  return (
    <h2>
      Student ID: {id}, Name: {searchParams.get('name') || 'Not provided'}
    </h2>
  );
}

export default function App() {
  return (
    <div>
      <h1>React Router Practical</h1>
      <nav>
        <Link to="/">Home</Link> |{' '}
        <Link to="/about">About</Link> |{' '}
        <Link to="/student/101?name=Amit">Student</Link>
      </nav>
      <Routes>
        <Route path="/" element={<Home />} />
        <Route path="/about" element={<About />} />
        <Route path="/student/:id" element={<Student />} />
      </Routes>
    </div>
  );
}

----- FILE: src/main.jsx -----

import React from 'react';
import { createRoot } from 'react-dom/client';
import { BrowserRouter } from 'react-router-dom';
import App from './App.jsx';

createRoot(document.getElementById('root')).render(
  <BrowserRouter><App /></BrowserRouter>
);

----- RUN -----
npm run dev
